async function loadDashboard(){ const content = document.getElementById('appContent'); let scriptElement = null; let timeoutId = null; let callbackName = null; function cleanup(){ if(timeoutId){ clearTimeout(timeoutId); timeoutId = null; } if(scriptElement){ scriptElement.remove(); scriptElement = null; } if(callbackName){ try{ delete window[callbackName]; }catch(error){ window[callbackName] = undefined; } callbackName = null; } } function showError(message){ cleanup(); content.innerHTML = `
خطا در دریافت اطلاعات داشبورد
${escapeHtml(message)}
`; } try{ if(!tg){ throw new Error( 'این داشبورد باید از داخل Telegram باز شود.' ); } const initData = tg.initData || ''; if(!initData){ throw new Error( 'Telegram initData دریافت نشد. داشبورد را از داخل ربات باز کنید.' ); } callbackName = '__dashboardCallback_' + Date.now() + '_' + Math.floor(Math.random() * 100000); const requestUrl = API_URL + '?action=getDashboard' + '&key=' + encodeURIComponent(API_KEY) + '&initData=' + encodeURIComponent(initData) + '&callback=' + encodeURIComponent(callbackName) + '&_=' + Date.now(); const resultHandler = function(result){ cleanup(); if(!result || result.ok !== true){ showError( result?.error || result?.message || 'دریافت اطلاعات داشبورد ناموفق بود.' ); return; } if(!result.data){ showError( 'اطلاعات داشبورد خالی است.' ); return; } renderDashboard(result.data); }; window[callbackName] = resultHandler; scriptElement = document.createElement('script'); scriptElement.src = requestUrl; scriptElement.async = true; scriptElement.onerror = function(){ showError( 'ارتباط با سرور داشبورد برقرار نشد. لطفاً دوباره تلاش کنید.' ); }; timeoutId = setTimeout(function(){ showError( 'پاسخ سرور بیش از حد طول کشید. لطفاً دوباره تلاش کنید.' ); }, 20000); document.head.appendChild(scriptElement); }catch(error){ console.error( 'Dashboard loading error:', error ); showError( error?.message || String(error) ); } }
